package org.delcom.ifs25048springbootcashflow.framework.auth;


import com.vaadin.flow.server.VaadinResponse;
import com.vaadin.flow.server.VaadinService;
import com.vaadin.flow.server.VaadinSession;
import com.vaadin.flow.server.WrappedSession;
import jakarta.servlet.http.Cookie;
import org.delcom.ifs25048springbootcashflow.domain.entity.User;
import org.delcom.ifs25048springbootcashflow.domain.repository.IUserRepository;

import java.util.Optional;

/**
 * Menyimpan sesi login di VaadinSession, HTTP session, dan cookie
 * agar tetap ada setelah refresh browser.
 */
public final class AuthSession {
    private static final String USER_KEY = "currentUser";
    private static final String USER_ID_KEY = "currentUserId";
    private static final String LOGGED_OUT_KEY = "authLoggedOut";
    private static final String COOKIE_NAME = "cashflow_uid";
    private static final int COOKIE_MAX_AGE_SECONDS = 60 * 60 * 24 * 7;

    private AuthSession() {
    }

    public static void login(User user) {
        VaadinSession session = VaadinSession.getCurrent();
        if (session == null) {
            return;
        }
        session.setAttribute(LOGGED_OUT_KEY, Boolean.FALSE);
        session.setAttribute(USER_KEY, user);
        session.setAttribute(USER_ID_KEY, user.getId());

        WrappedSession httpSession = session.getSession();
        if (httpSession != null) {
            httpSession.setAttribute(USER_ID_KEY, user.getId());
            httpSession.setAttribute(LOGGED_OUT_KEY, Boolean.FALSE);
        }
        writeUserCookie(user.getId());
    }

    public static void logout() {
        VaadinSession session = VaadinSession.getCurrent();
        if (session != null) {
            session.setAttribute(USER_KEY, null);
            session.setAttribute(USER_ID_KEY, null);
            session.setAttribute(LOGGED_OUT_KEY, Boolean.TRUE);
            WrappedSession httpSession = session.getSession();
            if (httpSession != null) {
                httpSession.setAttribute(USER_ID_KEY, null);
                httpSession.setAttribute(LOGGED_OUT_KEY, Boolean.TRUE);
            }
        }
        clearUserCookie();
    }

    public static Optional<User> currentUser() {
        VaadinSession session = VaadinSession.getCurrent();
        if (session == null || isLoggedOut(session)) {
            return Optional.empty();
        }
        Object value = session.getAttribute(USER_KEY);
        if (value instanceof User user) {
            return Optional.of(user);
        }
        return Optional.empty();
    }

    /**
     * Mengambil user login; jika hilang karena refresh, pulihkan dari session/cookie + database.
     */
    public static Optional<User> currentUser(IUserRepository userRepository) {
        VaadinSession session = VaadinSession.getCurrent();
        if (session != null && isLoggedOut(session)) {
            return Optional.empty();
        }
        Optional<User> existing = currentUser();
        if (existing.isPresent()) {
            return existing;
        }
        return restore(userRepository);
    }

    public static void refresh(User user) {
        login(user);
    }

    private static boolean isLoggedOut(VaadinSession session) {
        Object flag = session.getAttribute(LOGGED_OUT_KEY);
        if (Boolean.TRUE.equals(flag)) {
            return true;
        }
        WrappedSession httpSession = session.getSession();
        if (httpSession != null) {
            return Boolean.TRUE.equals(httpSession.getAttribute(LOGGED_OUT_KEY));
        }
        return false;
    }

    private static Optional<User> restore(IUserRepository userRepository) {
        VaadinSession session = VaadinSession.getCurrent();
        if (session != null && isLoggedOut(session)) {
            return Optional.empty();
        }
        Integer userId = readStoredUserId();
        if (userId == null) {
            return Optional.empty();
        }
        Optional<User> found = userRepository.findById(userId);
        found.ifPresent(AuthSession::login);
        if (found.isEmpty()) {
            clearUserCookie();
        }
        return found;
    }

    private static Integer readStoredUserId() {
        VaadinSession session = VaadinSession.getCurrent();
        if (session != null) {
            if (isLoggedOut(session)) {
                return null;
            }
            Object fromVaadin = session.getAttribute(USER_ID_KEY);
            if (fromVaadin instanceof Integer id) {
                return id;
            }
            WrappedSession httpSession = session.getSession();
            if (httpSession != null) {
                Object fromHttp = httpSession.getAttribute(USER_ID_KEY);
                if (fromHttp instanceof Integer id) {
                    return id;
                }
            }
        }
        return null;
    }

    private static void writeUserCookie(int userId) {
        VaadinResponse response = VaadinService.getCurrentResponse();
        if (response == null) {
            return;
        }
        Cookie cookie = new Cookie(COOKIE_NAME, "1");
        cookie.setPath("/");
        cookie.setHttpOnly(true);
        cookie.setMaxAge(COOKIE_MAX_AGE_SECONDS);
        response.addCookie(cookie);
    }

    private static void clearUserCookie() {
        VaadinResponse response = VaadinService.getCurrentResponse();
        if (response == null) {
            return;
        }
        Cookie cookie = new Cookie(COOKIE_NAME, "");
        cookie.setPath("/");
        cookie.setHttpOnly(true);
        cookie.setMaxAge(0);
        response.addCookie(cookie);
    }
}

    private void openAddDialog(TransactionType type) {

        Dialog dialog = new Dialog();

        TextArea description =
                new TextArea("Keterangan");

        TextField amount =
                new TextField("Jumlah (Rp)");

        description.setWidthFull();
        amount.setWidthFull();

        description.setRequiredIndicatorVisible(true);
        amount.setRequiredIndicatorVisible(true);

        amount.setPlaceholder("Contoh: 25000");

        AtomicReference<PendingImage> pending =
                new AtomicReference<>();

        Upload upload = new Upload(
                UploadHandler.inMemory((metadata, data) -> {

                    pending.set(new PendingImage(
                            metadata.fileName(),
                            data
                    ));
                })
        );

        upload.setAcceptedMimeTypes(
                "image/png",
                "image/jpeg"
        );

        upload.setAcceptedFileExtensions(
                ".png", ".jpg", ".jpeg"
        );

        upload.setMaxFiles(1);
        upload.setMaxFileSize(5 * 1024 * 1024);

        upload.addFileRemovedListener(event ->
                pending.set(null)
        );

        ProgressBar loading = new ProgressBar();
        loading.setIndeterminate(true);
        loading.setVisible(false);

        Button cancel = new Button(
                "Batal",
                event -> dialog.close()
        );

        Button save = new Button("Simpan");

        save.addThemeVariants(ButtonVariant.LUMO_PRIMARY);

        save.addClickListener(event -> {

            loading.setVisible(true);
            save.setEnabled(false);

            try {

                long value;

                try {

                    value = Long.parseLong(
                            amount.getValue().trim()
                    );

                } catch (NumberFormatException ex) {

                    throw new IllegalArgumentException(
                            "Jumlah harus berupa angka bulat"
                    );
                }

                if (value <= 0) {
                    throw new IllegalArgumentException(
                            "Jumlah harus lebih besar dari nol"
                    );
                }

                if (description.getValue().isBlank()) {
                    throw new IllegalArgumentException(
                            "Keterangan tidak boleh kosong"
                    );
                }

                String proofPath = null;

                PendingImage image = pending.get();

                if (image != null) {

                    String name = image.name()
                            .toLowerCase(Locale.ROOT);

                            || name.endsWith(".jpeg"))) {

                        throw new IllegalArgumentException(
                                "Format gambar tidak didukung"
                        );
                    }

                    if (image.data().length > 5 * 1024 * 1024) {

                        throw new IllegalArgumentException(
                                "Ukuran gambar maksimal 5 MB"
                        );
                    }

                    if (javax.imageio.ImageIO.read(
                            new ByteArrayInputStream(
                                    image.data()
                            )
                    ) == null) {

                        throw new IllegalArgumentException(
                                "File bukan gambar yang valid"
                        );
                    }

                    proofPath = fileStorage.store(
                            "proofs",
                            image.name(),
                            new ByteArrayInputStream(
                                    image.data()
                            )
                    );
                }

                Transaction saved =
                        financeUseCase.addTransaction(
                                currentUserId(),
                                description.getValue(),
                                value,
                                type,
                                proofPath
                        );

                success(presenter.successAdd(saved));

                dialog.close();

                refreshAll();

            } catch (Exception ex) {

                error(ex.getMessage());

            } finally {

                loading.setVisible(false);

        icon.getStyle().set("color", "#4B5563");

        text.getStyle()

        HorizontalLayout content = new HorizontalLayout(icon, text);
        content.setAlignItems(FlexComponent.Alignment.CENTER);
        content.setSpacing(true);
        content.setPadding(false);
        content.getStyle().set("gap", "0.65rem");

        RouterLink link = new RouterLink();
        link.setRoute(view);
        link.add(content);
        link.getStyle()
                .set("display", "block")
                .set("width", "100%")
                .set("text-decoration", "none")
                .set("padding", "0.7rem 0.85rem")
                .set("border-radius", "10px")
                .set("box-sizing", "border-box")
                .set("border", "1px solid transparent")
                .set("transition", "background 0.15s ease")
                .set("cursor", "pointer");

        link.getElement().addEventListener("mouseover", e -> {
            if (!"true".equals(link.getElement().getAttribute("data-active"))) {
                link.getStyle().set("background", "#F3F4F6");
            }
        });
        link.getElement().addEventListener("mouseout", e -> {
            if (!"true".equals(link.getElement().getAttribute("data-active"))) {
                link.getStyle().set("background", "transparent");
            }
        });

        navEntries.add(new NavEntry(link, pathPrefix));
        return link;
    }

    private void highlightActive(String path) {
        String normalized = path == null ? "" : path;
        if (normalized.startsWith("/")) {
            normalized = normalized.substring(1);
        }

        for (NavEntry entry : navEntries) {
            boolean active;
            if (entry.pathPrefix().isEmpty()) {
                active = normalized.isEmpty();
            } else {
                active = normalized.equals(entry.pathPrefix())
                        || normalized.startsWith(entry.pathPrefix() + "/");
            }
            // Highlight Daftar Pengguna juga saat melihat todo user lain.
            if ("users".equals(entry.pathPrefix()) && normalized.startsWith("user-transactions")) {
                active = true;
            }

            RouterLink link = entry.link();
            if (active) {
                link.getElement().setAttribute("data-active", "true");
                link.getStyle()
                        .set("background", "#DBEAFE")
                        .set("border", "1px solid #93C5FD");
                styleNavChildren(link, "#1D4ED8", "#1E40AF", "600");
            } else {
                link.getElement().setAttribute("data-active", "false");
                link.getStyle()
                        .set("background", "transparent")
                        .set("border", "1px solid transparent");
                styleNavChildren(link, "#4B5563", "#1F2937", "500");
            }
        }
    }

    private void styleNavChildren(RouterLink link, String iconColor, String textColor, String weight) {
        link.getChildren()
                .filter(HorizontalLayout.class::isInstance)
                .map(HorizontalLayout.class::cast)
                .findFirst()
                .ifPresent(row -> row.getChildren().forEach(child -> {
                    if (child instanceof Icon icon) {
                        icon.getStyle().set("color", iconColor);
                    }
                    if (child instanceof Span span) {
                        span.getStyle().set("color", textColor).set("font-weight", weight);
                    }
                }));
    }

    @Override
    public void afterNavigation(AfterNavigationEvent event) {
        highlightActive(event.getLocation().getPath());
        refreshHeaderAvatar(AuthSession.currentUser(userRepository).orElse(null));
    }

    @Override
    public void beforeEnter(BeforeEnterEvent event) {
        if (AuthSession.currentUser(userRepository).isEmpty()) {
            event.forwardTo(LoginView.class);
        }
    }

/**
 * Card konten halaman agar tampilan konsisten.
 * Komponen terakhir (biasanya grid) mengisi sisa tinggi agar tabel tidak terdesak.
 */
final class PageCard {

    private PageCard() {
    }

    static Div wrap(Component... content) {
        return wrap(true, content);
    }

    static Div wrap(boolean fullHeight, Component... content) {
        VerticalLayout body = new VerticalLayout();
        body.setPadding(false);
        body.setSpacing(false);
        body.setWidthFull();
        body.getStyle().set("gap", "0.65rem");

        for (Component component : content) {
            body.add(component);
        }

        if (fullHeight) {
            body.setSizeFull();
            body.getStyle().set("min-height", "0").set("flex", "1 1 auto");
            if (content.length > 0) {
                Component last = content[content.length - 1];
                body.setFlexGrow(1, last);
                last.getElement().getStyle().set("min-height", "0");
            }
        }

        Div card = new Div(body);
        card.setWidthFull();
        if (fullHeight) {
            card.setHeightFull();
            card.getStyle()
                    .set("display", "flex")
                    .set("flex-direction", "column")
                    .set("min-height", "0")
                    .set("overflow", "hidden");
        }
        card.getStyle()
                .set("background", "#FFFFFF")
                .set("border", "1px solid #E5E7EB")
                .set("border-radius", "12px")
                .set("padding", "1rem 1.1rem")
                .set("box-shadow", "0 8px 24px rgba(15, 23, 42, 0.06)")

import com.vaadin.flow.component.grid.Grid;

import com.vaadin.flow.router.BeforeEnterEvent;
import com.vaadin.flow.router.BeforeEnterObserver;
import com.vaadin.flow.router.BeforeEvent;
import com.vaadin.flow.router.HasUrlParameter;
import com.vaadin.flow.router.OptionalParameter;
import com.vaadin.flow.router.PageTitle;
import com.vaadin.flow.router.Route;

import org.delcom.ifs25048springbootcashflow.adapter.presenter.FinancePresenter;
import org.delcom.ifs25048springbootcashflow.domain.entity.Transaction;
import org.delcom.ifs25048springbootcashflow.domain.entity.User;

import org.delcom.ifs25048springbootcashflow.framework.auth.AuthSession;
import org.delcom.ifs25048springbootcashflow.usecase.FinanceUseCase;
import org.delcom.ifs25048springbootcashflow.usecase.UserUseCase;

@Route(value = "user-transactions", layout = MainLayout.class)
@PageTitle("Transaksi Pengguna")
public class UserTransactionsView
        extends VerticalLayout
        implements HasUrlParameter<Integer>,
                   BeforeEnterObserver {

    private final UserUseCase userUseCase;
    private final FinanceUseCase financeUseCase;
    private final FinancePresenter presenter;
    private final IUserRepository userRepository;

    private final H1 title =
            new H1("Transaksi Pengguna");

    private final Paragraph summary =
            new Paragraph();

    private final Grid<Transaction> grid =
            new Grid<>(Transaction.class, false);

    private Integer targetUserId;

    public UserTransactionsView(
            UserUseCase userUseCase,
            FinanceUseCase financeUseCase,
            FinancePresenter presenter,
            IUserRepository userRepository
    ) {

        this.userUseCase = userUseCase;
        this.financeUseCase = financeUseCase;
        this.presenter = presenter;
        this.userRepository = userRepository;

        setSizeFull();
        setPadding(true);

        getStyle().set("background", "#F3F4F6");

        configureGrid();

        Div card = PageCard.wrap(
                title,
                new Span("Mode read-only"),
                summary,
                grid
        );

        add(card);

        expand(card);
    }

    private void configureGrid() {

        grid.addComponentColumn(transaction -> {

            String path = transaction.getProofPath();

            if (path == null || path.isBlank()) {
                return new Span("-");
            }

            return ImagePreviewDialog.clickableThumbnail(
                    "/uploads/" + path,
                    "Bukti transaksi",
                    "48px"
            );

        }).setHeader("Bukti").setAutoWidth(true);

        grid.addColumn(Transaction::getId)
                .setHeader("ID");

        grid.addColumn(Transaction::getDescription)
                .setHeader("Keterangan")
                .setFlexGrow(2);

        grid.addColumn(transaction ->
                presenter.formatAmount(
                        transaction.getAmount()
                )
        ).setHeader("Jumlah");

        grid.addColumn(transaction ->
                transaction.getType().label()
        ).setHeader("Tipe");

        grid.addColumn(transaction ->
                presenter.formatDateTime(
                        transaction.getCreatedAt()
                )
        ).setHeader("Waktu");

        grid.setSizeFull();

        GridStyles.apply(grid);
    }

    @Override
    public void setParameter(
            BeforeEvent event,
            @OptionalParameter Integer userId
    ) {

        targetUserId = userId;
    }

    @Override
    public void beforeEnter(BeforeEnterEvent event) {

        User current = AuthSession
                .currentUser(userRepository)
                .orElse(null);

        if (current == null) {

            event.forwardTo(LoginView.class);
            return;
        }

        if (!current.isAdmin()) {

            event.forwardTo(FinanceView.class);
            return;
        }

        if (targetUserId == null || targetUserId <= 0) {

            event.forwardTo(UsersView.class);
            return;
        }

        try {

            User owner =
                    userUseCase.getUser(targetUserId);

            title.setText(
                    "Transaksi milik "
                    + owner.getFullName()
                    + " (@" + owner.getUsername() + ")"
            );

            grid.setItems(
                    financeUseCase.getTransactionsByUser(
                            targetUserId
                    )
            );

            summary.setText(
                    "Pemasukan: "
                    + presenter.formatAmount(
                        financeUseCase.getTotalIncome(targetUserId)
                    )
                    + " | Pengeluaran: "
                    + presenter.formatAmount(
                        financeUseCase.getTotalExpense(targetUserId)
                    )
                    + " | Saldo: "
                    + presenter.formatAmount(
                        financeUseCase.getBalance(targetUserId)
                    )
            );

        } catch (EntityNotFoundException ex) {

            event.forwardTo(UsersView.class);
        }
    }
}
